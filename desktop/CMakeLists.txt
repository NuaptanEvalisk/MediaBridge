cmake_minimum_required(VERSION 3.20)
project(MediaBridgeDesktop VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Network Concurrent)
qt_standard_project_setup()
qt_add_executable(mediabridge-desktop main.cpp branding.qrc)
target_link_libraries(mediabridge-desktop PRIVATE Qt6::Core Qt6::Widgets Qt6::Network Qt6::Concurrent)
install(TARGETS mediabridge-desktop RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  add_executable(mediabridge-adaptive-test tests/adaptive_parallelism_test.cpp)
  add_test(NAME adaptive_parallelism COMMAND mediabridge-adaptive-test)
  add_executable(mediabridge-cache-test tests/adaptive_cache_test.cpp)
  target_link_libraries(mediabridge-cache-test PRIVATE Qt6::Core)
  add_test(NAME adaptive_cache COMMAND mediabridge-cache-test)
  add_executable(mediabridge-charts-test tests/transfer_charts_test.cpp)
  target_link_libraries(mediabridge-charts-test PRIVATE Qt6::Widgets)
  add_test(NAME transfer_charts COMMAND mediabridge-charts-test)
  set_tests_properties(transfer_charts PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
